# Runtime dependencies for the bench poller.
# The server (server.py) and the agent clients (submit.py, run_md.py)
# are pure-stdlib -- they don't need anything from this file.
#
# *** EVERY LINE BELOW IS OPTIONAL. ***
# The poller try-imports each driver lazily; a bench that doesn't use
# a given device skips the matching install and the corresponding
# plugin's probe just returns "no devices." So you only need the
# packages your bench actually uses.

# UART -- mp135, dsp, bench_mcu, fpga (channel B), ssh probe path.
pyserial>=3.5

# FTDI MPSSE / FT2232H -- fpga (flash programmer).
ftd2xx>=1.3.3

# FTDI FT4222 SPI -- dsp (SPI link to the eval board).
# Distributed on PyPI as `pyft4222`; imported in code as `ft4222`.
pyft4222>=1.10

# libusb device enumeration -- _usb.winusb_device_present, used by
# msc and dfu probe paths to confirm the bootloader is enumerated.
pyusb>=1.2

# VISA / SCPI -- scope (oscilloscope capture).
pyvisa>=1.13
pyvisa-py>=0.7

# Plain WebSocket client -- ws plugin network stream tests.
websocket-client>=1.7

# Trace processing -- scope (downsampling + signal analysis).
numpy>=1.23
